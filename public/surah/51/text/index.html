بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ

  1. ١وَٱلذَّٰرِيَٰتِ ذَرۡوًا
  2. ٢فَٱلۡحَٰمِلَٰتِ وِقۡرًا
  3. ٣فَٱلۡجَٰرِيَٰتِ يُسۡرًا
  4. ٤فَٱلۡمُقَسِّمَٰتِ أَمۡرًا
  5. ٥إِنَّمَا تُوعَدُونَ لَصَادِقٌ
  6. ٦وَإِنَّ ٱلدِّينَ لَوَٰقِعٌ
  7. ٧وَٱلسَّمَآءِ ذَاتِ ٱلۡحُبُكِ
  8. ٨إِنَّكُمۡ لَفِى قَوۡلٍ مُّخۡتَلِفٍ
  9. ٩يُؤۡفَكُ عَنۡهُ مَنۡ أُفِكَ
  10. ١٠قُتِلَ ٱلۡخَرَّٰصُونَ
  11. ١١ٱلَّذِينَ هُمۡ فِى غَمۡرَةٍ سَاهُونَ
  12. ١٢يَسۡـَٔلُونَ أَيَّانَ يَوۡمُ ٱلدِّينِ
  13. ١٣يَوۡمَ هُمۡ عَلَى ٱلنَّارِ يُفۡتَنُونَ
  14. ١٤ذُوقُواْ فِتۡنَتَكُمۡ هَٰذَا ٱلَّذِى كُنتُم بِهِۦ تَسۡتَعۡجِلُونَ
  15. ١٥إِنَّ ٱلۡمُتَّقِينَ فِى جَنَّٰتٍ وَعُيُونٍ
  16. ١٦ءَاخِذِينَ مَآ ءَاتَىٰهُمۡ رَبُّهُمۡ إِنَّهُمۡ كَانُواْ قَبۡلَ ذَٰلِكَ مُحۡسِنِينَ
  17. ١٧كَانُواْ قَلِيلًا مِّنَ ٱلَّيۡلِ مَا يَهۡجَعُونَ
  18. ١٨وَبِٱلۡأَسۡحَارِ هُمۡ يَسۡتَغۡفِرُونَ
  19. ١٩وَفِىٓ أَمۡوَٰلِهِمۡ حَقٌّ لِّلسَّآئِلِ وَٱلۡمَحۡرُومِ
  20. ٢٠وَفِى ٱلۡأَرۡضِ ءَايَٰتٌ لِّلۡمُوقِنِينَ
  21. ٢١وَفِىٓ أَنفُسِكُمۡ أَفَلَا تُبۡصِرُونَ
  22. ٢٢وَفِى ٱلسَّمَآءِ رِزۡقُكُمۡ وَمَا تُوعَدُونَ
  23. ٢٣فَوَرَبِّ ٱلسَّمَآءِ وَٱلۡأَرۡضِ إِنَّهُۥ لَحَقٌّ مِّثۡلَ مَآ أَنَّكُمۡ تَنطِقُونَ
  24. ٢٤هَلۡ أَتَىٰكَ حَدِيثُ ضَيۡفِ إِبۡرَٰهِيمَ ٱلۡمُكۡرَمِينَ
  25. ٢٥إِذۡ دَخَلُواْ عَلَيۡهِ فَقَالُواْ سَلَٰمًا قَالَ سَلَٰمٌ قَوۡمٌ مُّنكَرُونَ
  26. ٢٦فَرَاغَ إِلَىٰٓ أَهۡلِهِۦ فَجَآءَ بِعِجۡلٍ سَمِينٍ
  27. ٢٧فَقَرَّبَهُۥٓ إِلَيۡهِمۡ قَالَ أَلَا تَأۡكُلُونَ
  28. ٢٨فَأَوۡجَسَ مِنۡهُمۡ خِيفَةً قَالُواْ لَا تَخَفۡ وَبَشَّرُوهُ بِغُلَٰمٍ عَلِيمٍ
  29. ٢٩فَأَقۡبَلَتِ ٱمۡرَأَتُهُۥ فِى صَرَّةٍ فَصَكَّتۡ وَجۡهَهَا وَقَالَتۡ عَجُوزٌ عَقِيمٌ
  30. ٣٠قَالُواْ كَذَٰلِكِ قَالَ رَبُّكِ إِنَّهُۥ هُوَ ٱلۡحَكِيمُ ٱلۡعَلِيمُ
  31. ٣١قَالَ فَمَا خَطۡبُكُمۡ أَيُّهَا ٱلۡمُرۡسَلُونَ
  32. ٣٢قَالُوٓاْ إِنَّآ أُرۡسِلۡنَآ إِلَىٰ قَوۡمٍ مُّجۡرِمِينَ
  33. ٣٣لِنُرۡسِلَ عَلَيۡهِمۡ حِجَارَةً مِّن طِينٍ
  34. ٣٤مُّسَوَّمَةً عِندَ رَبِّكَ لِلۡمُسۡرِفِينَ
  35. ٣٥فَأَخۡرَجۡنَا مَن كَانَ فِيهَا مِنَ ٱلۡمُؤۡمِنِينَ
  36. ٣٦فَمَا وَجَدۡنَا فِيهَا غَيۡرَ بَيۡتٍ مِّنَ ٱلۡمُسۡلِمِينَ
  37. ٣٧وَتَرَكۡنَا فِيهَآ ءَايَةً لِّلَّذِينَ يَخَافُونَ ٱلۡعَذَابَ ٱلۡأَلِيمَ
  38. ٣٨وَفِى مُوسَىٰٓ إِذۡ أَرۡسَلۡنَٰهُ إِلَىٰ فِرۡعَوۡنَ بِسُلۡطَٰنٍ مُّبِينٍ
  39. ٣٩فَتَوَلَّىٰ بِرُكۡنِهِۦ وَقَالَ سَٰحِرٌ أَوۡ مَجۡنُونٌ
  40. ٤٠فَأَخَذۡنَٰهُ وَجُنُودَهُۥ فَنَبَذۡنَٰهُمۡ فِى ٱلۡيَمِّ وَهُوَ مُلِيمٌ
  41. ٤١وَفِى عَادٍ إِذۡ أَرۡسَلۡنَا عَلَيۡهِمُ ٱلرِّيحَ ٱلۡعَقِيمَ
  42. ٤٢مَا تَذَرُ مِن شَىۡءٍ أَتَتۡ عَلَيۡهِ إِلَّا جَعَلَتۡهُ كَٱلرَّمِيمِ
  43. ٤٣وَفِى ثَمُودَ إِذۡ قِيلَ لَهُمۡ تَمَتَّعُواْ حَتَّىٰ حِينٍ
  44. ٤٤فَعَتَوۡاْ عَنۡ أَمۡرِ رَبِّهِمۡ فَأَخَذَتۡهُمُ ٱلصَّٰعِقَةُ وَهُمۡ يَنظُرُونَ
  45. ٤٥فَمَا ٱسۡتَطَٰعُواْ مِن قِيَامٍ وَمَا كَانُواْ مُنتَصِرِينَ
  46. ٤٦وَقَوۡمَ نُوحٍ مِّن قَبۡلُ إِنَّهُمۡ كَانُواْ قَوۡمًا فَٰسِقِينَ
  47. ٤٧وَٱلسَّمَآءَ بَنَيۡنَٰهَا بِأَيۡيْدٍ وَإِنَّا لَمُوسِعُونَ
  48. ٤٨وَٱلۡأَرۡضَ فَرَشۡنَٰهَا فَنِعۡمَ ٱلۡمَٰهِدُونَ
  49. ٤٩وَمِن كُلِّ شَىۡءٍ خَلَقۡنَا زَوۡجَيۡنِ لَعَلَّكُمۡ تَذَكَّرُونَ
  50. ٥٠فَفِرُّوٓاْ إِلَى ٱللَّهِ إِنِّى لَكُم مِّنۡهُ نَذِيرٌ مُّبِينٌ
  51. ٥١وَلَا تَجۡعَلُواْ مَعَ ٱللَّهِ إِلَٰهًا ءَاخَرَ إِنِّى لَكُم مِّنۡهُ نَذِيرٌ مُّبِينٌ
  52. ٥٢كَذَٰلِكَ مَآ أَتَى ٱلَّذِينَ مِن قَبۡلِهِم مِّن رَّسُولٍ إِلَّا قَالُواْ سَاحِرٌ أَوۡ مَجۡنُونٌ
  53. ٥٣أَتَوَاصَوۡاْ بِهِۦ بَلۡ هُمۡ قَوۡمٌ طَاغُونَ
  54. ٥٤فَتَوَلَّ عَنۡهُمۡ فَمَآ أَنتَ بِمَلُومٍ
  55. ٥٥وَذَكِّرۡ فَإِنَّ ٱلذِّكۡرَىٰ تَنفَعُ ٱلۡمُؤۡمِنِينَ
  56. ٥٦وَمَا خَلَقۡتُ ٱلۡجِنَّ وَٱلۡإِنسَ إِلَّا لِيَعۡبُدُونِ
  57. ٥٧مَآ أُرِيدُ مِنۡهُم مِّن رِّزۡقٍ وَمَآ أُرِيدُ أَن يُطۡعِمُونِ
  58. ٥٨إِنَّ ٱللَّهَ هُوَ ٱلرَّزَّاقُ ذُو ٱلۡقُوَّةِ ٱلۡمَتِينُ
  59. ٥٩فَإِنَّ لِلَّذِينَ ظَلَمُواْ ذَنُوبًا مِّثۡلَ ذَنُوبِ أَصۡحَٰبِهِمۡ فَلَا يَسۡتَعۡجِلُونِ
  60. ٦٠فَوَيۡلٌ لِّلَّذِينَ كَفَرُواْ مِن يَوۡمِهِمُ ٱلَّذِى يُوعَدُونَ

سُورَةُ غَافِرٍ — 85 آية · 85 verses · الصفحات 467–476 · pages 467–476

بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ

  1. ١حمٓ
  2. ٢تَنزِيلُ ٱلۡكِتَٰبِ مِنَ ٱللَّهِ ٱلۡعَزِيزِ ٱلۡعَلِيمِ
  3. ٣غَافِرِ ٱلذَّنۢبِ وَقَابِلِ ٱلتَّوۡبِ شَدِيدِ ٱلۡعِقَابِ ذِى ٱلطَّوۡلِ لَآ إِلَٰهَ إِلَّا هُوَ إِلَيۡهِ ٱلۡمَصِيرُ
  4. ٤مَا يُجَٰدِلُ فِىٓ ءَايَٰتِ ٱللَّهِ إِلَّا ٱلَّذِينَ كَفَرُواْ فَلَا يَغۡرُرۡكَ تَقَلُّبُهُمۡ فِى ٱلۡبِلَٰدِ
  5. ٥كَذَّبَتۡ قَبۡلَهُمۡ قَوۡمُ نُوحٍ وَٱلۡأَحۡزَابُ مِنۢ بَعۡدِهِمۡ وَهَمَّتۡ كُلُّ أُمَّةٍۭ بِرَسُولِهِمۡ لِيَأۡخُذُوهُ وَجَٰدَلُواْ بِٱلۡبَٰطِلِ لِيُدۡحِضُواْ بِهِ ٱلۡحَقَّ فَأَخَذۡتُهُمۡ فَكَيۡفَ كَانَ عِقَابِ
  6. ٦وَكَذَٰلِكَ حَقَّتۡ كَلِمَتُ رَبِّكَ عَلَى ٱلَّذِينَ كَفَرُوٓاْ أَنَّهُمۡ أَصۡحَٰبُ ٱلنَّارِ
  7. ٧ٱلَّذِينَ يَحۡمِلُونَ ٱلۡعَرۡشَ وَمَنۡ حَوۡلَهُۥ يُسَبِّحُونَ بِحَمۡدِ رَبِّهِمۡ وَيُؤۡمِنُونَ بِهِۦ وَيَسۡتَغۡفِرُونَ لِلَّذِينَ ءَامَنُواْ رَبَّنَا وَسِعۡتَ كُلَّ شَىۡءٍ رَّحۡمَةً وَعِلۡمًا فَٱغۡفِرۡ لِلَّذِينَ تَابُواْ وَٱتَّبَعُواْ سَبِيلَكَ وَقِهِمۡ عَذَابَ ٱلۡجَحِيمِ
  8. ٨رَبَّنَا وَأَدۡخِلۡهُمۡ جَنَّٰتِ عَدۡنٍ ٱلَّتِى وَعَدتَّهُمۡ وَمَن صَلَحَ مِنۡ ءَابَآئِهِمۡ وَأَزۡوَٰجِهِمۡ وَذُرِّيَّٰتِهِمۡ إِنَّكَ أَنتَ ٱلۡعَزِيزُ ٱلۡحَكِيمُ
  9. ٩وَقِهِمُ ٱلسَّيِّـَٔاتِ وَمَن تَقِ ٱلسَّيِّـَٔاتِ يَوۡمَئِذٍ فَقَدۡ رَحِمۡتَهُۥ وَذَٰلِكَ هُوَ ٱلۡفَوۡزُ ٱلۡعَظِيمُ
  10. ١٠إِنَّ ٱلَّذِينَ كَفَرُواْ يُنَادَوۡنَ لَمَقۡتُ ٱللَّهِ أَكۡبَرُ مِن مَّقۡتِكُمۡ أَنفُسَكُمۡ إِذۡ تُدۡعَوۡنَ إِلَى ٱلۡإِيمَٰنِ فَتَكۡفُرُونَ
  11. ١١قَالُواْ رَبَّنَآ أَمَتَّنَا ٱثۡنَتَيۡنِ وَأَحۡيَيۡتَنَا ٱثۡنَتَيۡنِ فَٱعۡتَرَفۡنَا بِذُنُوبِنَا فَهَلۡ إِلَىٰ خُرُوجٍ مِّن سَبِيلٍ
  12. ١٢ذَٰلِكُم بِأَنَّهُۥٓ إِذَا دُعِىَ ٱللَّهُ وَحۡدَهُۥ كَفَرۡتُمۡ وَإِن يُشۡرَكۡ بِهِۦ تُؤۡمِنُواْ فَٱلۡحُكۡمُ لِلَّهِ ٱلۡعَلِىِّ ٱلۡكَبِيرِ
  13. ١٣هُوَ ٱلَّذِى يُرِيكُمۡ ءَايَٰتِهِۦ وَيُنَزِّلُ لَكُم مِّنَ ٱلسَّمَآءِ رِزۡقًا وَمَا يَتَذَكَّرُ إِلَّا مَن يُنِيبُ
  14. ١٤فَٱدۡعُواْ ٱللَّهَ مُخۡلِصِينَ لَهُ ٱلدِّينَ وَلَوۡ كَرِهَ ٱلۡكَٰفِرُونَ
  15. ١٥رَفِيعُ ٱلدَّرَجَٰتِ ذُو ٱلۡعَرۡشِ يُلۡقِى ٱلرُّوحَ مِنۡ أَمۡرِهِۦ عَلَىٰ مَن يَشَآءُ مِنۡ عِبَادِهِۦ لِيُنذِرَ يَوۡمَ ٱلتَّلَاقِ
  16. ١٦يَوۡمَ هُم بَٰرِزُونَ لَا يَخۡفَىٰ عَلَى ٱللَّهِ مِنۡهُمۡ شَىۡءٌ لِّمَنِ ٱلۡمُلۡكُ ٱلۡيَوۡمَ لِلَّهِ ٱلۡوَٰحِدِ ٱلۡقَهَّارِ
  17. ١٧ٱلۡيَوۡمَ تُجۡزَىٰ كُلُّ نَفۡسٍۭ بِمَا كَسَبَتۡ لَا ظُلۡمَ ٱلۡيَوۡمَ إِنَّ ٱللَّهَ سَرِيعُ ٱلۡحِسَابِ
  18. ١٨وَأَنذِرۡهُمۡ يَوۡمَ ٱلۡـَٔازِفَةِ إِذِ ٱلۡقُلُوبُ لَدَى ٱلۡحَنَاجِرِ كَٰظِمِينَ مَا لِلظَّٰلِمِينَ مِنۡ حَمِيمٍ وَلَا شَفِيعٍ يُطَاعُ
  19. ١٩يَعۡلَمُ خَآئِنَةَ ٱلۡأَعۡيُنِ وَمَا تُخۡفِى ٱلصُّدُورُ
  20. ٢٠وَٱللَّهُ يَقۡضِى بِٱلۡحَقِّ وَٱلَّذِينَ يَدۡعُونَ مِن دُونِهِۦ لَا يَقۡضُونَ بِشَىۡءٍ إِنَّ ٱللَّهَ هُوَ ٱلسَّمِيعُ ٱلۡبَصِيرُ
  21. ٢١أَوَلَمۡ يَسِيرُواْ فِى ٱلۡأَرۡضِ فَيَنظُرُواْ كَيۡفَ كَانَ عَٰقِبَةُ ٱلَّذِينَ كَانُواْ مِن قَبۡلِهِمۡ كَانُواْ هُمۡ أَشَدَّ مِنۡهُمۡ قُوَّةً وَءَاثَارًا فِى ٱلۡأَرۡضِ فَأَخَذَهُمُ ٱللَّهُ بِذُنُوبِهِمۡ وَمَا كَانَ لَهُم مِّنَ ٱللَّهِ مِن وَاقٍ
  22. ٢٢ذَٰلِكَ بِأَنَّهُمۡ كَانَت تَّأۡتِيهِمۡ رُسُلُهُم بِٱلۡبَيِّنَٰتِ فَكَفَرُواْ فَأَخَذَهُمُ ٱللَّهُ إِنَّهُۥ قَوِىٌّ شَدِيدُ ٱلۡعِقَابِ
  23. ٢٣وَلَقَدۡ أَرۡسَلۡنَا مُوسَىٰ بِـَٔايَٰتِنَا وَسُلۡطَٰنٍ مُّبِينٍ
  24. ٢٤إِلَىٰ فِرۡعَوۡنَ وَهَٰمَٰنَ وَقَٰرُونَ فَقَالُواْ سَٰحِرٌ كَذَّابٌ
  25. ٢٥فَلَمَّا جَآءَهُم بِٱلۡحَقِّ مِنۡ عِندِنَا قَالُواْ ٱقۡتُلُوٓاْ أَبۡنَآءَ ٱلَّذِينَ ءَامَنُواْ مَعَهُۥ وَٱسۡتَحۡيُواْ نِسَآءَهُمۡ وَمَا كَيۡدُ ٱلۡكَٰفِرِينَ إِلَّا فِى ضَلَٰلٍ
  26. ٢٦وَقَالَ فِرۡعَوۡنُ ذَرُونِىٓ أَقۡتُلۡ مُوسَىٰ وَلۡيَدۡعُ رَبَّهُۥٓ إِنِّىٓ أَخَافُ أَن يُبَدِّلَ دِينَكُمۡ أَوۡ أَن يُظۡهِرَ فِى ٱلۡأَرۡضِ ٱلۡفَسَادَ
  27. ٢٧وَقَالَ مُوسَىٰٓ إِنِّى عُذۡتُ بِرَبِّى وَرَبِّكُم مِّن كُلِّ مُتَكَبِّرٍ لَّا يُؤۡمِنُ بِيَوۡمِ ٱلۡحِسَابِ
  28. ٢٨وَقَالَ رَجُلٌ مُّؤۡمِنٌ مِّنۡ ءَالِ فِرۡعَوۡنَ يَكۡتُمُ إِيمَٰنَهُۥٓ أَتَقۡتُلُونَ رَجُلًا أَن يَقُولَ رَبِّىَ ٱللَّهُ وَقَدۡ جَآءَكُم بِٱلۡبَيِّنَٰتِ مِن رَّبِّكُمۡ وَإِن يَكُ كَٰذِبًا فَعَلَيۡهِ كَذِبُهُۥ وَإِن يَكُ صَادِقًا يُصِبۡكُم بَعۡضُ ٱلَّذِى يَعِدُكُمۡ إِنَّ ٱللَّهَ لَا يَهۡدِى مَنۡ هُوَ مُسۡرِفٌ كَذَّابٌ
  29. ٢٩يَٰقَوۡمِ لَكُمُ ٱلۡمُلۡكُ ٱلۡيَوۡمَ ظَٰهِرِينَ فِى ٱلۡأَرۡضِ فَمَن يَنصُرُنَا مِنۢ بَأۡسِ ٱللَّهِ إِن جَآءَنَا قَالَ فِرۡعَوۡنُ مَآ أُرِيكُمۡ إِلَّا مَآ أَرَىٰ وَمَآ أَهۡدِيكُمۡ إِلَّا سَبِيلَ ٱلرَّشَادِ
  30. ٣٠وَقَالَ ٱلَّذِىٓ ءَامَنَ يَٰقَوۡمِ إِنِّىٓ أَخَافُ عَلَيۡكُم مِّثۡلَ يَوۡمِ ٱلۡأَحۡزَابِ
  31. ٣١مِثۡلَ دَأۡبِ قَوۡمِ نُوحٍ وَعَادٍ وَثَمُودَ وَٱلَّذِينَ مِنۢ بَعۡدِهِمۡ وَمَا ٱللَّهُ يُرِيدُ ظُلۡمًا لِّلۡعِبَادِ
  32. ٣٢وَيَٰقَوۡمِ إِنِّىٓ أَخَافُ عَلَيۡكُمۡ يَوۡمَ ٱلتَّنَادِ
  33. ٣٣يَوۡمَ تُوَلُّونَ مُدۡبِرِينَ مَا لَكُم مِّنَ ٱللَّهِ مِنۡ عَاصِمٍ وَمَن يُضۡلِلِ ٱللَّهُ فَمَا لَهُۥ مِنۡ هَادٍ
  34. ٣٤وَلَقَدۡ جَآءَكُمۡ يُوسُفُ مِن قَبۡلُ بِٱلۡبَيِّنَٰتِ فَمَا زِلۡتُمۡ فِى شَكٍّ مِّمَّا جَآءَكُم بِهِۦ حَتَّىٰٓ إِذَا هَلَكَ قُلۡتُمۡ لَن يَبۡعَثَ ٱللَّهُ مِنۢ بَعۡدِهِۦ رَسُولًا كَذَٰلِكَ يُضِلُّ ٱللَّهُ مَنۡ هُوَ مُسۡرِفٌ مُّرۡتَابٌ
  35. ٣٥ٱلَّذِينَ يُجَٰدِلُونَ فِىٓ ءَايَٰتِ ٱللَّهِ بِغَيۡرِ سُلۡطَٰنٍ أَتَىٰهُمۡ كَبُرَ مَقۡتًا عِندَ ٱللَّهِ وَعِندَ ٱلَّذِينَ ءَامَنُواْ كَذَٰلِكَ يَطۡبَعُ ٱللَّهُ عَلَىٰ كُلِّ قَلۡبِ مُتَكَبِّرٍ جَبَّارٍ
  36. ٣٦وَقَالَ فِرۡعَوۡنُ يَٰهَٰمَٰنُ ٱبۡنِ لِى صَرۡحًا لَّعَلِّىٓ أَبۡلُغُ ٱلۡأَسۡبَٰبَ
  37. ٣٧أَسۡبَٰبَ ٱلسَّمَٰوَٰتِ فَأَطَّلِعَ إِلَىٰٓ إِلَٰهِ مُوسَىٰ وَإِنِّى لَأَظُنُّهُۥ كَٰذِبًا وَكَذَٰلِكَ زُيِّنَ لِفِرۡعَوۡنَ سُوٓءُ عَمَلِهِۦ وَصُدَّ عَنِ ٱلسَّبِيلِ وَمَا كَيۡدُ فِرۡعَوۡنَ إِلَّا فِى تَبَابٍ
  38. ٣٨وَقَالَ ٱلَّذِىٓ ءَامَنَ يَٰقَوۡمِ ٱتَّبِعُونِ أَهۡدِكُمۡ سَبِيلَ ٱلرَّشَادِ
  39. ٣٩يَٰقَوۡمِ إِنَّمَا هَٰذِهِ ٱلۡحَيَوٰةُ ٱلدُّنۡيَا مَتَٰعٌ وَإِنَّ ٱلۡـَٔاخِرَةَ هِىَ دَارُ ٱلۡقَرَارِ
  40. ٤٠مَنۡ عَمِلَ سَيِّئَةً فَلَا يُجۡزَىٰٓ إِلَّا مِثۡلَهَا وَمَنۡ عَمِلَ صَٰلِحًا مِّن ذَكَرٍ أَوۡ أُنثَىٰ وَهُوَ مُؤۡمِنٌ فَأُوْلَٰٓئِكَ يَدۡخُلُونَ ٱلۡجَنَّةَ يُرۡزَقُونَ فِيهَا بِغَيۡرِ حِسَابٍ
  41. ٤١وَيَٰقَوۡمِ مَا لِىٓ أَدۡعُوكُمۡ إِلَى ٱلنَّجَوٰةِ وَتَدۡعُونَنِىٓ إِلَى ٱلنَّارِ
  42. ٤٢تَدۡعُونَنِى لِأَكۡفُرَ بِٱللَّهِ وَأُشۡرِكَ بِهِۦ مَا لَيۡسَ لِى بِهِۦ عِلۡمٌ وَأَنَا۠ أَدۡعُوكُمۡ إِلَى ٱلۡعَزِيزِ ٱلۡغَفَّٰرِ
  43. ٤٣لَا جَرَمَ أَنَّمَا تَدۡعُونَنِىٓ إِلَيۡهِ لَيۡسَ لَهُۥ دَعۡوَةٌ فِى ٱلدُّنۡيَا وَلَا فِى ٱلۡـَٔاخِرَةِ وَأَنَّ مَرَدَّنَآ إِلَى ٱللَّهِ وَأَنَّ ٱلۡمُسۡرِفِينَ هُمۡ أَصۡحَٰبُ ٱلنَّارِ
  44. ٤٤فَسَتَذۡكُرُونَ مَآ أَقُولُ لَكُمۡ وَأُفَوِّضُ أَمۡرِىٓ إِلَى ٱللَّهِ إِنَّ ٱللَّهَ بَصِيرٌۢ بِٱلۡعِبَادِ
  45. ٤٥فَوَقَىٰهُ ٱللَّهُ سَيِّـَٔاتِ مَا مَكَرُواْ وَحَاقَ بِـَٔالِ فِرۡعَوۡنَ سُوٓءُ ٱلۡعَذَابِ
  46. ٤٦ٱلنَّارُ يُعۡرَضُونَ عَلَيۡهَا غُدُوًّا وَعَشِيًّا وَيَوۡمَ تَقُومُ ٱلسَّاعَةُ أَدۡخِلُوٓاْ ءَالَ فِرۡعَوۡنَ أَشَدَّ ٱلۡعَذَابِ
  47. ٤٧وَإِذۡ يَتَحَآجُّونَ فِى ٱلنَّارِ فَيَقُولُ ٱلضُّعَفَٰٓؤُاْ لِلَّذِينَ ٱسۡتَكۡبَرُوٓاْ إِنَّا كُنَّا لَكُمۡ تَبَعًا فَهَلۡ أَنتُم مُّغۡنُونَ عَنَّا نَصِيبًا مِّنَ ٱلنَّارِ
  48. ٤٨قَالَ ٱلَّذِينَ ٱسۡتَكۡبَرُوٓاْ إِنَّا كُلٌّ فِيهَآ إِنَّ ٱللَّهَ قَدۡ حَكَمَ بَيۡنَ ٱلۡعِبَادِ
  49. ٤٩وَقَالَ ٱلَّذِينَ فِى ٱلنَّارِ لِخَزَنَةِ جَهَنَّمَ ٱدۡعُواْ رَبَّكُمۡ يُخَفِّفۡ عَنَّا يَوۡمًا مِّنَ ٱلۡعَذَابِ
  50. ٥٠قَالُوٓاْ أَوَلَمۡ تَكُ تَأۡتِيكُمۡ رُسُلُكُم بِٱلۡبَيِّنَٰتِ قَالُواْ بَلَىٰ قَالُواْ فَٱدۡعُواْ وَمَا دُعَٰٓؤُاْ ٱلۡكَٰفِرِينَ إِلَّا فِى ضَلَٰلٍ
  51. ٥١إِنَّا لَنَنصُرُ رُسُلَنَا وَٱلَّذِينَ ءَامَنُواْ فِى ٱلۡحَيَوٰةِ ٱلدُّنۡيَا وَيَوۡمَ يَقُومُ ٱلۡأَشۡهَٰدُ
  52. ٥٢يَوۡمَ لَا يَنفَعُ ٱلظَّٰلِمِينَ مَعۡذِرَتُهُمۡ وَلَهُمُ ٱللَّعۡنَةُ وَلَهُمۡ سُوٓءُ ٱلدَّارِ
  53. ٥٣وَلَقَدۡ ءَاتَيۡنَا مُوسَى ٱلۡهُدَىٰ وَأَوۡرَثۡنَا بَنِىٓ إِسۡرَٰٓءِيلَ ٱلۡكِتَٰبَ
  54. ٥٤هُدًى وَذِكۡرَىٰ لِأُوْلِى ٱلۡأَلۡبَٰبِ
  55. ٥٥فَٱصۡبِرۡ إِنَّ وَعۡدَ ٱللَّهِ حَقٌّ وَٱسۡتَغۡفِرۡ لِذَنۢبِكَ وَسَبِّحۡ بِحَمۡدِ رَبِّكَ بِٱلۡعَشِىِّ وَٱلۡإِبۡكَٰرِ
  56. ٥٦إِنَّ ٱلَّذِينَ يُجَٰدِلُونَ فِىٓ ءَايَٰتِ ٱللَّهِ بِغَيۡرِ سُلۡطَٰنٍ أَتَىٰهُمۡ إِن فِى صُدُورِهِمۡ إِلَّا كِبۡرٌ مَّا هُم بِبَٰلِغِيهِ فَٱسۡتَعِذۡ بِٱللَّهِ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلۡبَصِيرُ
  57. ٥٧لَخَلۡقُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ أَكۡبَرُ مِنۡ خَلۡقِ ٱلنَّاسِ وَلَٰكِنَّ أَكۡثَرَ ٱلنَّاسِ لَا يَعۡلَمُونَ
  58. ٥٨وَمَا يَسۡتَوِى ٱلۡأَعۡمَىٰ وَٱلۡبَصِيرُ وَٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّٰلِحَٰتِ وَلَا ٱلۡمُسِىٓءُ قَلِيلًا مَّا تَتَذَكَّرُونَ
  59. ٥٩إِنَّ ٱلسَّاعَةَ لَـَٔاتِيَةٌ لَّا رَيۡبَ فِيهَا وَلَٰكِنَّ أَكۡثَرَ ٱلنَّاسِ لَا يُؤۡمِنُونَ
  60. ٦٠وَقَالَ رَبُّكُمُ ٱدۡعُونِىٓ أَسۡتَجِبۡ لَكُمۡ إِنَّ ٱلَّذِينَ يَسۡتَكۡبِرُونَ عَنۡ عِبَادَتِى سَيَدۡخُلُونَ جَهَنَّمَ دَاخِرِينَ
  61. ٦١ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلَّيۡلَ لِتَسۡكُنُواْ فِيهِ وَٱلنَّهَارَ مُبۡصِرًا إِنَّ ٱللَّهَ لَذُو فَضۡلٍ عَلَى ٱلنَّاسِ وَلَٰكِنَّ أَكۡثَرَ ٱلنَّاسِ لَا يَشۡكُرُونَ
  62. ٦٢ذَٰلِكُمُ ٱللَّهُ رَبُّكُمۡ خَٰلِقُ كُلِّ شَىۡءٍ لَّآ إِلَٰهَ إِلَّا هُوَ فَأَنَّىٰ تُؤۡفَكُونَ
  63. ٦٣كَذَٰلِكَ يُؤۡفَكُ ٱلَّذِينَ كَانُواْ بِـَٔايَٰتِ ٱللَّهِ يَجۡحَدُونَ
  64. ٦٤ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلۡأَرۡضَ قَرَارًا وَٱلسَّمَآءَ بِنَآءً وَصَوَّرَكُمۡ فَأَحۡسَنَ صُوَرَكُمۡ وَرَزَقَكُم مِّنَ ٱلطَّيِّبَٰتِ ذَٰلِكُمُ ٱللَّهُ رَبُّكُمۡ فَتَبَارَكَ ٱللَّهُ رَبُّ ٱلۡعَٰلَمِينَ
  65. ٦٥هُوَ ٱلۡحَىُّ لَآ إِلَٰهَ إِلَّا هُوَ فَٱدۡعُوهُ مُخۡلِصِينَ لَهُ ٱلدِّينَ ٱلۡحَمۡدُ لِلَّهِ رَبِّ ٱلۡعَٰلَمِينَ
  66. ٦٦قُلۡ إِنِّى نُهِيتُ أَنۡ أَعۡبُدَ ٱلَّذِينَ تَدۡعُونَ مِن دُونِ ٱللَّهِ لَمَّا جَآءَنِىَ ٱلۡبَيِّنَٰتُ مِن رَّبِّى وَأُمِرۡتُ أَنۡ أُسۡلِمَ لِرَبِّ ٱلۡعَٰلَمِينَ
  67. ٦٧هُوَ ٱلَّذِى خَلَقَكُم مِّن تُرَابٍ ثُمَّ مِن نُّطۡفَةٍ ثُمَّ مِنۡ عَلَقَةٍ ثُمَّ يُخۡرِجُكُمۡ طِفۡلًا ثُمَّ لِتَبۡلُغُوٓاْ أَشُدَّكُمۡ ثُمَّ لِتَكُونُواْ شُيُوخًا وَمِنكُم مَّن يُتَوَفَّىٰ مِن قَبۡلُ وَلِتَبۡلُغُوٓاْ أَجَلًا مُّسَمًّى وَلَعَلَّكُمۡ تَعۡقِلُونَ
  68. ٦٨هُوَ ٱلَّذِى يُحۡىِۦ وَيُمِيتُ فَإِذَا قَضَىٰٓ أَمۡرًا فَإِنَّمَا يَقُولُ لَهُۥ كُن فَيَكُونُ
  69. ٦٩أَلَمۡ تَرَ إِلَى ٱلَّذِينَ يُجَٰدِلُونَ فِىٓ ءَايَٰتِ ٱللَّهِ أَنَّىٰ يُصۡرَفُونَ
  70. ٧٠ٱلَّذِينَ كَذَّبُواْ بِٱلۡكِتَٰبِ وَبِمَآ أَرۡسَلۡنَا بِهِۦ رُسُلَنَا فَسَوۡفَ يَعۡلَمُونَ
  71. ٧١إِذِ ٱلۡأَغۡلَٰلُ فِىٓ أَعۡنَٰقِهِمۡ وَٱلسَّلَٰسِلُ يُسۡحَبُونَ
  72. ٧٢فِى ٱلۡحَمِيمِ ثُمَّ فِى ٱلنَّارِ يُسۡجَرُونَ
  73. ٧٣ثُمَّ قِيلَ لَهُمۡ أَيۡنَ مَا كُنتُمۡ تُشۡرِكُونَ
  74. ٧٤مِن دُونِ ٱللَّهِ قَالُواْ ضَلُّواْ عَنَّا بَل لَّمۡ نَكُن نَّدۡعُواْ مِن قَبۡلُ شَيۡـًٔا كَذَٰلِكَ يُضِلُّ ٱللَّهُ ٱلۡكَٰفِرِينَ
  75. ٧٥ذَٰلِكُم بِمَا كُنتُمۡ تَفۡرَحُونَ فِى ٱلۡأَرۡضِ بِغَيۡرِ ٱلۡحَقِّ وَبِمَا كُنتُمۡ تَمۡرَحُونَ
  76. ٧٦ٱدۡخُلُوٓاْ أَبۡوَٰبَ جَهَنَّمَ خَٰلِدِينَ فِيهَا فَبِئۡسَ مَثۡوَى ٱلۡمُتَكَبِّرِينَ
  77. ٧٧فَٱصۡبِرۡ إِنَّ وَعۡدَ ٱللَّهِ حَقٌّ فَإِمَّا نُرِيَنَّكَ بَعۡضَ ٱلَّذِى نَعِدُهُمۡ أَوۡ نَتَوَفَّيَنَّكَ فَإِلَيۡنَا يُرۡجَعُونَ
  78. ٧٨وَلَقَدۡ أَرۡسَلۡنَا رُسُلًا مِّن قَبۡلِكَ مِنۡهُم مَّن قَصَصۡنَا عَلَيۡكَ وَمِنۡهُم مَّن لَّمۡ نَقۡصُصۡ عَلَيۡكَ وَمَا كَانَ لِرَسُولٍ أَن يَأۡتِىَ بِـَٔايَةٍ إِلَّا بِإِذۡنِ ٱللَّهِ فَإِذَا جَآءَ أَمۡرُ ٱللَّهِ قُضِىَ بِٱلۡحَقِّ وَخَسِرَ هُنَالِكَ ٱلۡمُبۡطِلُونَ
  79. ٧٩ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلۡأَنۡعَٰمَ لِتَرۡكَبُواْ مِنۡهَا وَمِنۡهَا تَأۡكُلُونَ
  80. ٨٠وَلَكُمۡ فِيهَا مَنَٰفِعُ وَلِتَبۡلُغُواْ عَلَيۡهَا حَاجَةً فِى صُدُورِكُمۡ وَعَلَيۡهَا وَعَلَى ٱلۡفُلۡكِ تُحۡمَلُونَ
  81. ٨١وَيُرِيكُمۡ ءَايَٰتِهِۦ فَأَىَّ ءَايَٰتِ ٱللَّهِ تُنكِرُونَ
  82. ٨٢أَفَلَمۡ يَسِيرُواْ فِى ٱلۡأَرۡضِ فَيَنظُرُواْ كَيۡفَ كَانَ عَٰقِبَةُ ٱلَّذِينَ مِن قَبۡلِهِمۡ كَانُوٓاْ أَكۡثَرَ مِنۡهُمۡ وَأَشَدَّ قُوَّةً وَءَاثَارًا فِى ٱلۡأَرۡضِ فَمَآ أَغۡنَىٰ عَنۡهُم مَّا كَانُواْ يَكۡسِبُونَ
  83. ٨٣فَلَمَّا جَآءَتۡهُمۡ رُسُلُهُم بِٱلۡبَيِّنَٰتِ فَرِحُواْ بِمَا عِندَهُم مِّنَ ٱلۡعِلۡمِ وَحَاقَ بِهِم مَّا كَانُواْ بِهِۦ يَسۡتَهۡزِءُونَ
  84. ٨٤فَلَمَّا رَأَوۡاْ بَأۡسَنَا قَالُوٓاْ ءَامَنَّا بِٱللَّهِ وَحۡدَهُۥ وَكَفَرۡنَا بِمَا كُنَّا بِهِۦ مُشۡرِكِينَ
  85. ٨٥فَلَمۡ يَكُ يَنفَعُهُمۡ إِيمَٰنُهُمۡ لَمَّا رَأَوۡاْ بَأۡسَنَا سُنَّتَ ٱللَّهِ ٱلَّتِى قَدۡ خَلَتۡ فِى عِبَادِهِۦ وَخَسِرَ هُنَالِكَ ٱلۡكَٰفِرُونَ

بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ

  1. ١يسٓ
  2. ٢وَٱلۡقُرۡءَانِ ٱلۡحَكِيمِ
  3. ٣إِنَّكَ لَمِنَ ٱلۡمُرۡسَلِينَ
  4. ٤عَلَىٰ صِرَٰطٍ مُّسۡتَقِيمٍ
  5. ٥تَنزِيلَ ٱلۡعَزِيزِ ٱلرَّحِيمِ
  6. ٦لِتُنذِرَ قَوۡمًا مَّآ أُنذِرَ ءَابَآؤُهُمۡ فَهُمۡ غَٰفِلُونَ
  7. ٧لَقَدۡ حَقَّ ٱلۡقَوۡلُ عَلَىٰٓ أَكۡثَرِهِمۡ فَهُمۡ لَا يُؤۡمِنُونَ
  8. ٨إِنَّا جَعَلۡنَا فِىٓ أَعۡنَٰقِهِمۡ أَغۡلَٰلًا فَهِىَ إِلَى ٱلۡأَذۡقَانِ فَهُم مُّقۡمَحُونَ
  9. ٩وَجَعَلۡنَا مِنۢ بَيۡنِ أَيۡدِيهِمۡ سَدًّا وَمِنۡ خَلۡفِهِمۡ سَدًّا فَأَغۡشَيۡنَٰهُمۡ فَهُمۡ لَا يُبۡصِرُونَ
  10. ١٠وَسَوَآءٌ عَلَيۡهِمۡ ءَأَنذَرۡتَهُمۡ أَمۡ لَمۡ تُنذِرۡهُمۡ لَا يُؤۡمِنُونَ
  11. ١١إِنَّمَا تُنذِرُ مَنِ ٱتَّبَعَ ٱلذِّكۡرَ وَخَشِىَ ٱلرَّحۡمَٰنَ بِٱلۡغَيۡبِ فَبَشِّرۡهُ بِمَغۡفِرَةٍ وَأَجۡرٍ كَرِيمٍ
  12. ١٢إِنَّا نَحۡنُ نُحۡىِ ٱلۡمَوۡتَىٰ وَنَكۡتُبُ مَا قَدَّمُواْ وَءَاثَٰرَهُمۡ وَكُلَّ شَىۡءٍ أَحۡصَيۡنَٰهُ فِىٓ إِمَامٍ مُّبِينٍ
  13. ١٣وَٱضۡرِبۡ لَهُم مَّثَلًا أَصۡحَٰبَ ٱلۡقَرۡيَةِ إِذۡ جَآءَهَا ٱلۡمُرۡسَلُونَ
  14. ١٤إِذۡ أَرۡسَلۡنَآ إِلَيۡهِمُ ٱثۡنَيۡنِ فَكَذَّبُوهُمَا فَعَزَّزۡنَا بِثَالِثٍ فَقَالُوٓاْ إِنَّآ إِلَيۡكُم مُّرۡسَلُونَ
  15. ١٥قَالُواْ مَآ أَنتُمۡ إِلَّا بَشَرٌ مِّثۡلُنَا وَمَآ أَنزَلَ ٱلرَّحۡمَٰنُ مِن شَىۡءٍ إِنۡ أَنتُمۡ إِلَّا تَكۡذِبُونَ
  16. ١٦قَالُواْ رَبُّنَا يَعۡلَمُ إِنَّآ إِلَيۡكُمۡ لَمُرۡسَلُونَ
  17. ١٧وَمَا عَلَيۡنَآ إِلَّا ٱلۡبَلَٰغُ ٱلۡمُبِينُ
  18. ١٨قَالُوٓاْ إِنَّا تَطَيَّرۡنَا بِكُمۡ لَئِن لَّمۡ تَنتَهُواْ لَنَرۡجُمَنَّكُمۡ وَلَيَمَسَّنَّكُم مِّنَّا عَذَابٌ أَلِيمٌ
  19. ١٩قَالُواْ طَٰٓئِرُكُم مَّعَكُمۡ أَئِن ذُكِّرۡتُم بَلۡ أَنتُمۡ قَوۡمٌ مُّسۡرِفُونَ
  20. ٢٠وَجَآءَ مِنۡ أَقۡصَا ٱلۡمَدِينَةِ رَجُلٌ يَسۡعَىٰ قَالَ يَٰقَوۡمِ ٱتَّبِعُواْ ٱلۡمُرۡسَلِينَ
  21. ٢١ٱتَّبِعُواْ مَن لَّا يَسۡـَٔلُكُمۡ أَجۡرًا وَهُم مُّهۡتَدُونَ
  22. ٢٢وَمَا لِىَ لَآ أَعۡبُدُ ٱلَّذِى فَطَرَنِى وَإِلَيۡهِ تُرۡجَعُونَ
  23. ٢٣ءَأَتَّخِذُ مِن دُونِهِۦٓ ءَالِهَةً إِن يُرِدۡنِ ٱلرَّحۡمَٰنُ بِضُرٍّ لَّا تُغۡنِ عَنِّى شَفَٰعَتُهُمۡ شَيۡـًٔا وَلَا يُنقِذُونِ
  24. ٢٤إِنِّىٓ إِذًا لَّفِى ضَلَٰلٍ مُّبِينٍ
  25. ٢٥إِنِّىٓ ءَامَنتُ بِرَبِّكُمۡ فَٱسۡمَعُونِ
  26. ٢٦قِيلَ ٱدۡخُلِ ٱلۡجَنَّةَ قَالَ يَٰلَيۡتَ قَوۡمِى يَعۡلَمُونَ
  27. ٢٧بِمَا غَفَرَ لِى رَبِّى وَجَعَلَنِى مِنَ ٱلۡمُكۡرَمِينَ
  28. ٢٨وَمَآ أَنزَلۡنَا عَلَىٰ قَوۡمِهِۦ مِنۢ بَعۡدِهِۦ مِن جُندٍ مِّنَ ٱلسَّمَآءِ وَمَا كُنَّا مُنزِلِينَ
  29. ٢٩إِن كَانَتۡ إِلَّا صَيۡحَةً وَٰحِدَةً فَإِذَا هُمۡ خَٰمِدُونَ
  30. ٣٠يَٰحَسۡرَةً عَلَى ٱلۡعِبَادِ مَا يَأۡتِيهِم مِّن رَّسُولٍ إِلَّا كَانُواْ بِهِۦ يَسۡتَهۡزِءُونَ
  31. ٣١أَلَمۡ يَرَوۡاْ كَمۡ أَهۡلَكۡنَا قَبۡلَهُم مِّنَ ٱلۡقُرُونِ أَنَّهُمۡ إِلَيۡهِمۡ لَا يَرۡجِعُونَ
  32. ٣٢وَإِن كُلٌّ لَّمَّا جَمِيعٌ لَّدَيۡنَا مُحۡضَرُونَ
  33. ٣٣وَءَايَةٌ لَّهُمُ ٱلۡأَرۡضُ ٱلۡمَيۡتَةُ أَحۡيَيۡنَٰهَا وَأَخۡرَجۡنَا مِنۡهَا حَبًّا فَمِنۡهُ يَأۡكُلُونَ
  34. ٣٤وَجَعَلۡنَا فِيهَا جَنَّٰتٍ مِّن نَّخِيلٍ وَأَعۡنَٰبٍ وَفَجَّرۡنَا فِيهَا مِنَ ٱلۡعُيُونِ
  35. ٣٥لِيَأۡكُلُواْ مِن ثَمَرِهِۦ وَمَا عَمِلَتۡهُ أَيۡدِيهِمۡ أَفَلَا يَشۡكُرُونَ
  36. ٣٦سُبۡحَٰنَ ٱلَّذِى خَلَقَ ٱلۡأَزۡوَٰجَ كُلَّهَا مِمَّا تُنۢبِتُ ٱلۡأَرۡضُ وَمِنۡ أَنفُسِهِمۡ وَمِمَّا لَا يَعۡلَمُونَ
  37. ٣٧وَءَايَةٌ لَّهُمُ ٱلَّيۡلُ نَسۡلَخُ مِنۡهُ ٱلنَّهَارَ فَإِذَا هُم مُّظۡلِمُونَ
  38. ٣٨وَٱلشَّمۡسُ تَجۡرِى لِمُسۡتَقَرٍّ لَّهَا ذَٰلِكَ تَقۡدِيرُ ٱلۡعَزِيزِ ٱلۡعَلِيمِ
  39. ٣٩وَٱلۡقَمَرَ قَدَّرۡنَٰهُ مَنَازِلَ حَتَّىٰ عَادَ كَٱلۡعُرۡجُونِ ٱلۡقَدِيمِ
  40. ٤٠لَا ٱلشَّمۡسُ يَنۢبَغِى لَهَآ أَن تُدۡرِكَ ٱلۡقَمَرَ وَلَا ٱلَّيۡلُ سَابِقُ ٱلنَّهَارِ وَكُلٌّ فِى فَلَكٍ يَسۡبَحُونَ
  41. ٤١وَءَايَةٌ لَّهُمۡ أَنَّا حَمَلۡنَا ذُرِّيَّتَهُمۡ فِى ٱلۡفُلۡكِ ٱلۡمَشۡحُونِ
  42. ٤٢وَخَلَقۡنَا لَهُم مِّن مِّثۡلِهِۦ مَا يَرۡكَبُونَ
  43. ٤٣وَإِن نَّشَأۡ نُغۡرِقۡهُمۡ فَلَا صَرِيخَ لَهُمۡ وَلَا هُمۡ يُنقَذُونَ
  44. ٤٤إِلَّا رَحۡمَةً مِّنَّا وَمَتَٰعًا إِلَىٰ حِينٍ
  45. ٤٥وَإِذَا قِيلَ لَهُمُ ٱتَّقُواْ مَا بَيۡنَ أَيۡدِيكُمۡ وَمَا خَلۡفَكُمۡ لَعَلَّكُمۡ تُرۡحَمُونَ
  46. ٤٦وَمَا تَأۡتِيهِم مِّنۡ ءَايَةٍ مِّنۡ ءَايَٰتِ رَبِّهِمۡ إِلَّا كَانُواْ عَنۡهَا مُعۡرِضِينَ
  47. ٤٧وَإِذَا قِيلَ لَهُمۡ أَنفِقُواْ مِمَّا رَزَقَكُمُ ٱللَّهُ قَالَ ٱلَّذِينَ كَفَرُواْ لِلَّذِينَ ءَامَنُوٓاْ أَنُطۡعِمُ مَن لَّوۡ يَشَآءُ ٱللَّهُ أَطۡعَمَهُۥٓ إِنۡ أَنتُمۡ إِلَّا فِى ضَلَٰلٍ مُّبِينٍ
  48. ٤٨وَيَقُولُونَ مَتَىٰ هَٰذَا ٱلۡوَعۡدُ إِن كُنتُمۡ صَٰدِقِينَ
  49. ٤٩مَا يَنظُرُونَ إِلَّا صَيۡحَةً وَٰحِدَةً تَأۡخُذُهُمۡ وَهُمۡ يَخِصِّمُونَ
  50. ٥٠فَلَا يَسۡتَطِيعُونَ تَوۡصِيَةً وَلَآ إِلَىٰٓ أَهۡلِهِمۡ يَرۡجِعُونَ
  51. ٥١وَنُفِخَ فِى ٱلصُّورِ فَإِذَا هُم مِّنَ ٱلۡأَجۡدَاثِ إِلَىٰ رَبِّهِمۡ يَنسِلُونَ
  52. ٥٢قَالُواْ يَٰوَيۡلَنَا مَنۢ بَعَثَنَا مِن مَّرۡقَدِنَا هَٰذَا مَا وَعَدَ ٱلرَّحۡمَٰنُ وَصَدَقَ ٱلۡمُرۡسَلُونَ
  53. ٥٣إِن كَانَتۡ إِلَّا صَيۡحَةً وَٰحِدَةً فَإِذَا هُمۡ جَمِيعٌ لَّدَيۡنَا مُحۡضَرُونَ
  54. ٥٤فَٱلۡيَوۡمَ لَا تُظۡلَمُ نَفۡسٌ شَيۡـًٔا وَلَا تُجۡزَوۡنَ إِلَّا مَا كُنتُمۡ تَعۡمَلُونَ
  55. ٥٥إِنَّ أَصۡحَٰبَ ٱلۡجَنَّةِ ٱلۡيَوۡمَ فِى شُغُلٍ فَٰكِهُونَ
  56. ٥٦هُمۡ وَأَزۡوَٰجُهُمۡ فِى ظِلَٰلٍ عَلَى ٱلۡأَرَآئِكِ مُتَّكِـُٔونَ
  57. ٥٧لَهُمۡ فِيهَا فَٰكِهَةٌ وَلَهُم مَّا يَدَّعُونَ
  58. ٥٨سَلَٰمٌ قَوۡلًا مِّن رَّبٍّ رَّحِيمٍ
  59. ٥٩وَٱمۡتَٰزُواْ ٱلۡيَوۡمَ أَيُّهَا ٱلۡمُجۡرِمُونَ
  60. ٦٠أَلَمۡ أَعۡهَدۡ إِلَيۡكُمۡ يَٰبَنِىٓ ءَادَمَ أَن لَّا تَعۡبُدُواْ ٱلشَّيۡطَٰنَ إِنَّهُۥ لَكُمۡ عَدُوٌّ مُّبِينٌ
  61. ٦١وَأَنِ ٱعۡبُدُونِى هَٰذَا صِرَٰطٌ مُّسۡتَقِيمٌ
  62. ٦٢وَلَقَدۡ أَضَلَّ مِنكُمۡ جِبِلًّا كَثِيرًا أَفَلَمۡ تَكُونُواْ تَعۡقِلُونَ
  63. ٦٣هَٰذِهِۦ جَهَنَّمُ ٱلَّتِى كُنتُمۡ تُوعَدُونَ
  64. ٦٤ٱصۡلَوۡهَا ٱلۡيَوۡمَ بِمَا كُنتُمۡ تَكۡفُرُونَ
  65. ٦٥ٱلۡيَوۡمَ نَخۡتِمُ عَلَىٰٓ أَفۡوَٰهِهِمۡ وَتُكَلِّمُنَآ أَيۡدِيهِمۡ وَتَشۡهَدُ أَرۡجُلُهُم بِمَا كَانُواْ يَكۡسِبُونَ
  66. ٦٦وَلَوۡ نَشَآءُ لَطَمَسۡنَا عَلَىٰٓ أَعۡيُنِهِمۡ فَٱسۡتَبَقُواْ ٱلصِّرَٰطَ فَأَنَّىٰ يُبۡصِرُونَ
  67. ٦٧وَلَوۡ نَشَآءُ لَمَسَخۡنَٰهُمۡ عَلَىٰ مَكَانَتِهِمۡ فَمَا ٱسۡتَطَٰعُواْ مُضِيًّا وَلَا يَرۡجِعُونَ
  68. ٦٨وَمَن نُّعَمِّرۡهُ نُنَكِّسۡهُ فِى ٱلۡخَلۡقِ أَفَلَا يَعۡقِلُونَ
  69. ٦٩وَمَا عَلَّمۡنَٰهُ ٱلشِّعۡرَ وَمَا يَنۢبَغِى لَهُۥٓ إِنۡ هُوَ إِلَّا ذِكۡرٌ وَقُرۡءَانٌ مُّبِينٌ
  70. ٧٠لِّيُنذِرَ مَن كَانَ حَيًّا وَيَحِقَّ ٱلۡقَوۡلُ عَلَى ٱلۡكَٰفِرِينَ
  71. ٧١أَوَلَمۡ يَرَوۡاْ أَنَّا خَلَقۡنَا لَهُم مِّمَّا عَمِلَتۡ أَيۡدِينَآ أَنۡعَٰمًا فَهُمۡ لَهَا مَٰلِكُونَ
  72. ٧٢وَذَلَّلۡنَٰهَا لَهُمۡ فَمِنۡهَا رَكُوبُهُمۡ وَمِنۡهَا يَأۡكُلُونَ
  73. ٧٣وَلَهُمۡ فِيهَا مَنَٰفِعُ وَمَشَارِبُ أَفَلَا يَشۡكُرُونَ
  74. ٧٤وَٱتَّخَذُواْ مِن دُونِ ٱللَّهِ ءَالِهَةً لَّعَلَّهُمۡ يُنصَرُونَ
  75. ٧٥لَا يَسۡتَطِيعُونَ نَصۡرَهُمۡ وَهُمۡ لَهُمۡ جُندٌ مُّحۡضَرُونَ
  76. ٧٦فَلَا يَحۡزُنكَ قَوۡلُهُمۡ إِنَّا نَعۡلَمُ مَا يُسِرُّونَ وَمَا يُعۡلِنُونَ
  77. ٧٧أَوَلَمۡ يَرَ ٱلۡإِنسَٰنُ أَنَّا خَلَقۡنَٰهُ مِن نُّطۡفَةٍ فَإِذَا هُوَ خَصِيمٌ مُّبِينٌ
  78. ٧٨وَضَرَبَ لَنَا مَثَلًا وَنَسِىَ خَلۡقَهُۥ قَالَ مَن يُحۡىِ ٱلۡعِظَٰمَ وَهِىَ رَمِيمٌ
  79. ٧٩قُلۡ يُحۡيِيهَا ٱلَّذِىٓ أَنشَأَهَآ أَوَّلَ مَرَّةٍ وَهُوَ بِكُلِّ خَلۡقٍ عَلِيمٌ
  80. ٨٠ٱلَّذِى جَعَلَ لَكُم مِّنَ ٱلشَّجَرِ ٱلۡأَخۡضَرِ نَارًا فَإِذَآ أَنتُم مِّنۡهُ تُوقِدُونَ
  81. ٨١أَوَلَيۡسَ ٱلَّذِى خَلَقَ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَ بِقَٰدِرٍ عَلَىٰٓ أَن يَخۡلُقَ مِثۡلَهُم بَلَىٰ وَهُوَ ٱلۡخَلَّٰقُ ٱلۡعَلِيمُ
  82. ٨٢إِنَّمَآ أَمۡرُهُۥٓ إِذَآ أَرَادَ شَيۡـًٔا أَن يَقُولَ لَهُۥ كُن فَيَكُونُ
  83. ٨٣فَسُبۡحَٰنَ ٱلَّذِى بِيَدِهِۦ مَلَكُوتُ كُلِّ شَىۡءٍ وَإِلَيۡهِ تُرۡجَعُونَ
